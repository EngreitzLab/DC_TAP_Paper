Compare Effect Sizes Across with paper

manual_pairs_path <- here("metadata", "validation_30_pairs.tsv")
manual_pairs_df <- read.csv(manual_pairs_path, sep = "\t", header=TRUE, stringsAsFactors = FALSE)

table_s3_path <- here("..", "results", "formatted_dc_tap_results", "Table_S3_Final_DC_TAP_Seq_Results_w_Chromatin_Categories_on_resized_and_merged_elements_250908_append.tsv")
table_s3 <- fread(table_s3_path, data.table = FALSE)

table_s3_wtc11 <- table_s3[table_s3$cell_type == "WTC11", ]
table_s3_wtc11 <- table_s3_wtc11[table_s3_wtc11$element_gene_pair_identifier_hg38 %in% manual_pairs_df$element_gene_pair_identifier_hg38, ]
# Put NA to false
table_s3_wtc11[is.na(table_s3_wtc11$significant_wo_pos_controls_20fdr), "significant_wo_pos_controls_20fdr"] <- FALSE

validation_pairs <- table_s3_wtc11
validation_ids <- unique(validation_pairs$element_gene_pair_identifier_hg38)
cat("Validation pairs:", length(validation_ids), "\n")
## Validation pairs: 30
cat("Validation pairs (Table S3 significant):", sum(validation_pairs$significant_wo_pos_controls_20fdr), "\n")
## Validation pairs (Table S3 significant): 24
union_dirs <- c(
    "MOI1_1_union",
    "MOI1_5_union",
    "MOI11_3_union",
    "MOI13_union",
    "low_moi_merged_union",
    "high_moi_merged_union"
)

sceptre_list <- lapply(union_dirs, function(dir_name) {
    fp <- file.path(sceptre_dir, "runs", dir_name, "sceptre_discovery_results.tsv")
    x <- fread(fp, data.table = FALSE)
    x$source <- dir_name
    x
})
sceptre_all <- bind_rows(sceptre_list)

gencode_v32_path <- here("data", "gencode.v32.primary_assembly.annotation.gtf.gz")
gencode_genes <- fread(
    cmd = paste("gzip -dc", shQuote(gencode_v32_path)),
    sep = "\t",
    header = FALSE,
    quote = "",
    data.table = FALSE,
    fill = TRUE,
    comment.char = "#"
)
colnames(gencode_genes) <- c(
    "seqname", "source", "feature", "start", "end", "score",
    "strand", "frame", "attribute"
)
gencode_genes <- gencode_genes[gencode_genes$feature == "gene", "attribute", drop = FALSE]
gencode_genes$gene_id <- sub('.*gene_id "([^"]+)".*', '\\1', gencode_genes$attribute)
gencode_genes$gene_symbol <- sub('.*gene_name "([^"]+)".*', '\\1', gencode_genes$attribute)
gencode_genes$gene_id_stripped <- sub("\\..*$", "", gencode_genes$gene_id)
gencode_genes <- unique(gencode_genes[, c("gene_id_stripped", "gene_symbol")])

sceptre_all$response_id_stripped <- sub("\\..*$", "", sceptre_all$response_id)
sceptre_all <- merge(
    sceptre_all,
    gencode_genes,
    by.x = "response_id_stripped",
    by.y = "gene_id_stripped",
    all.x = TRUE,
    sort = FALSE
)

sceptre_all$pct_change_effect_size <- 100 * (sceptre_all$fold_change - 1)
sceptre_all$se_pct_change_effect_size <- 100 * sceptre_all$se_fold_change
sceptre_all$element_gene_pair_identifier_hg38 <- paste(
    sceptre_all$gene_symbol,
    sceptre_all$grna_target,
    sep = "|"
)

Compare the paper with high-moi merged

# Keep only relevant columns from the paper results
paper_effects <- validation_pairs %>%
  select(
    element_gene_pair_identifier_hg38,
    gene_symbol,
    pct_change_effect_size_paper = pct_change_effect_size,
    se_pct_change_effect_size_paper = standard_error_pct_change,
    significant_paper = significant_wo_pos_controls_20fdr
  )

# Join with sceptre results
plot_df <- sceptre_all %>%
  select(
    element_gene_pair_identifier_hg38,
    gene_symbol,
    source,
    pct_change_effect_size_sceptre = pct_change_effect_size,
    se_pct_change_effect_size_sceptre = se_pct_change_effect_size,
    significant_sceptre = significant
  ) %>%
  inner_join(
    paper_effects,
    by = c("element_gene_pair_identifier_hg38", "gene_symbol")
  )

plot_df <- plot_df %>%
  filter(source == "high_moi_merged_union")

cor.value <- cor(plot_df$pct_change_effect_size_paper, plot_df$pct_change_effect_size_sceptre, method = "pearson")
cor.text <- paste("r =", round(cor.value, 2))

# Scatter plot with error bars
ggplot(
  plot_df,
  aes(
    x = pct_change_effect_size_paper,
    y = pct_change_effect_size_sceptre,
  )
) +

  # horizontal error bars
  geom_errorbar(
    aes(
      xmin = pct_change_effect_size_paper - 1.96 *
        se_pct_change_effect_size_paper,
      xmax = pct_change_effect_size_paper + 1.96 *
        se_pct_change_effect_size_paper
    ),
    orientation = "y",
    width = 0
  ) +

  # vertical error bars
  geom_errorbar(
    aes(
      ymin = pct_change_effect_size_sceptre - 1.96 *
        se_pct_change_effect_size_sceptre,
      ymax = pct_change_effect_size_sceptre + 1.96 *
        se_pct_change_effect_size_sceptre
    ),
    width = 0
  ) +

  geom_point(alpha = 0.6, size = 0.1) +

  geom_abline(
    slope = 1,
    intercept = 0,
    linetype = "dotted",
    color = "black"
  ) +

  ylim(-50, 10) +
  geom_vline(xintercept = 0, linetype = "dashed") +
  
  geom_hline(yintercept = 0, linetype = "dashed") +

  annotate("text",
           x = -Inf, y = Inf,
           label = cor.text,
           hjust = -0.1, vjust = 1.5,
           size = 4, color = "red") +

  theme_classic() +

  labs(
    x = "Full Screen Effect Size(%)",
    y = "High-Moi Effect Size(%)",
    color = "Significant",
    title = "Effect Size Concordance"
  ) +
  theme(
    aspect.ratio = 1
  ) + coord_flip()

Second panel for paper

Compare validation low-moi vs with high-moi merged

# Select sceptre results
plot_df <- sceptre_all %>%
  select(
    element_gene_pair_identifier_hg38,
    gene_symbol,
    source,
    pct_change_effect_size,
    se_pct_change_effect_size,
    significant_sceptre = significant
  ) %>%
  filter(source %in% c("high_moi_merged_union", "low_moi_merged_union")) %>%
  filter(element_gene_pair_identifier_hg38 %in% validation_pairs$element_gene_pair_identifier_hg38) %>%
  tidyr::pivot_wider(
    id_cols = c(element_gene_pair_identifier_hg38, gene_symbol),
    names_from = source,
    values_from = c(
      pct_change_effect_size,
      se_pct_change_effect_size,
      significant_sceptre
    )
  )

cor.value <- cor(
  plot_df$pct_change_effect_size_high_moi_merged_union,
  plot_df$pct_change_effect_size_low_moi_merged_union,
  method = "pearson",
  use = "complete.obs"
)

cor.text <- paste("r =", round(cor.value, 2))

# Scatter plot with error bars
ggplot(
  plot_df,
  aes(
    x = pct_change_effect_size_high_moi_merged_union,
    y = pct_change_effect_size_low_moi_merged_union,
    color = significant_sceptre_high_moi_merged_union |
      significant_sceptre_low_moi_merged_union
  )
) +
  geom_errorbar(
  aes(
    xmin = pct_change_effect_size_high_moi_merged_union -
      1.96 * se_pct_change_effect_size_high_moi_merged_union,
    xmax = pct_change_effect_size_high_moi_merged_union +
      1.96 * se_pct_change_effect_size_high_moi_merged_union
  ),
  orientation = "y",
  width = 0
  ) +
  geom_errorbar(
    aes(
      ymin = pct_change_effect_size_low_moi_merged_union -
        1.96 * se_pct_change_effect_size_low_moi_merged_union,
      ymax = pct_change_effect_size_low_moi_merged_union +
        1.96 * se_pct_change_effect_size_low_moi_merged_union
    ),
    width = 0
  ) +
  geom_point(alpha = 0.6, size = 0.8) +
  geom_abline(slope = 1, intercept = 0, linetype = "dotted") +
  geom_vline(xintercept = 0, linetype = "dashed") +
  geom_hline(yintercept = 0, linetype = "dashed") +
  coord_cartesian(xlim = c(-50, 10), ylim = c(-50, 10)) +
  annotate(
    "text",
    x = -Inf, y = Inf,
    label = cor.text,
    hjust = -0.1, vjust = 1.5,
    size = 4,
    color = "red"
  ) +
  scale_color_manual(values = c("TRUE" = "#d62728", "FALSE" = "grey70")) +
  theme_classic() +
  labs(
    x = "Validation Screen High-MOI Effect Size (%)",
    y = "Validation Screen Low-MOI Effect Size (%)",
    color = "Significant",
    title = "Effect Size Concordance"
  ) +
  theme(aspect.ratio = 1)

ggsave( filename = file.path(output_dir, "scatter_effect_size_correlation_high_vs_low.pdf"), width = 6, height = 6, dpi = 300 )
